name: Backend CI

on:
  workflow_dispatch:
  push:
    branches: [ main, development ]
  pull_request:
    branches: [ main, development ]

jobs:
  backend:
    permissions:
      contents: read
      pull-requests: write
    runs-on: ubuntu-latest
    defaults:
      run:
        working-directory: backend
    steps:
      - uses: actions/checkout@v4

      - name: Setup Node.js 22.19.0 LTS
        uses: actions/setup-node@v4
        with:
          node-version: 22.19.0

      - name: Install dependencies
        run: npm ci

      - name: Set up .env
        run: cp .env.example .env

      - name: Run typecheck
        run: npm run typecheck

      # -------------------------------
      # 1️⃣ Run backend server for smoke/integration
      # -------------------------------
      - name: Run backend server
        run: |
          PORT=3001 npm run dev > server.log 2>&1 &
          SERVER_PID=$!
          echo "Server PID: $SERVER_PID"
          for i in {1..15}; do
            if curl -s http://localhost:3001/health > /dev/null; then
              echo "Server is up!"
              break
            fi
            sleep 2
          done
          if ! curl -f http://localhost:3001/health > /dev/null; then
            echo "Server failed to start:"
            cat server.log
            kill $SERVER_PID
            exit 1
          fi
          echo "SERVER_PID=$SERVER_PID" >> $GITHUB_ENV

      # -------------------------------
      # 2️⃣ Run backend tests
      # -------------------------------
      - name: Run backend tests
        run: npm run test -- --run --reporter=dot

      # -------------------------------
      # 3️⃣ Stop backend server
      # -------------------------------
      - name: Stop backend server
        if: always()
        run: kill $SERVER_PID

      # -------------------------------
      # 📝 Comment in PR on failure
      # -------------------------------
      - name: Comment on PR if failed
        if: ${{ failure() && github.event_name == 'pull_request' }}
        uses: actions/github-script@v7
        with:
          script: |
            const pr = context.payload.pull_request?.number;
            if (!pr) {
              core.info('No PR number found; skipping comment.');
              return;
            }
            const body = [
              '**Backend CI** failed.',
              `Commit: \`${process.env.GITHUB_SHA.slice(0,7)}\``,
              `Logs: ${process.env.GITHUB_SERVER_URL}/${process.env.GITHUB_REPOSITORY}/actions/runs/${process.env.GITHUB_RUN_ID}`
            ].join('\n');
            await github.rest.issues.createComment({
              owner: context.repo.owner,
              repo: context.repo.repo,
              issue_number: pr,
              body
            });
